class PowerShellWorker(QThread):
    """Worker Thread für PowerShell-Befehle"""
    output_signal = pyqtSignal(str)
    error_signal = pyqtSignal(str)
    finished_signal = pyqtSignal(bool)
    
    def __init__(self, script):
        super().__init__()
        self.script = script
    
    def run(self):
        try:
            result = subprocess.run(
                ["powershell", "-ExecutionPolicy", "Bypass", "-Command", self.script],
                capture_output=True,
                text=True,
                timeout=300
            )
            
            if result.stdout:
                self.output_signal.emit(result.stdout)
            
            if result.returncode == 0:
                self.finished_signal.emit(True)
            else:
                self.error_signal.emit(result.stderr if result.stderr else "Unbekannter Fehler")
                self.finished_signal.emit(False)
                
        except Exception as e:
            self.error_signal.emit(str(e))
            self.finished_signal.emit(False)

    def refresh_vm_list(self):
        script = "Get-VM | Select-Object -ExpandProperty Name"
        worker = PowerShellWorker(script)
        worker.output_signal.connect(lambda output: self.populate_vm_combo(output))
        worker.error_signal.connect(self.log_output)
        worker.start()
    
    def populate_vm_combo(self, output):
        self.vm_select_combo.clear()
        vms = [vm.strip() for vm in output.strip().split('\n') if vm.strip()]
        self.vm_select_combo.addItems(vms)
    
    # VM-Erstellung
    def create_vm(self):
        vm_name = self.vm_name_input.text().strip()
        if not vm_name:
            QMessageBox.warning(self, "Fehler", "Bitte VM-Name eingeben!")
            return
        
        self.vm_name = vm_name
        ram_gb = self.ram_input.value()
        path = self.path_input.text()
        vhdx_size = self.vhdx_size_input.value()
        iso_path = self.iso_input.text()
        
        script = f"""
# Hyper-V-Dienst prüfen und starten
$service = Get-Service -Name vmms
if ($service.Status -ne 'Running') {{
    Start-Service -Name vmms
    Write-Output "Hyper-V-Dienst wurde gestartet."
}}

# VM erstellen
$VMPath = '{path}'
$VMName = '{vm_name}'
$RAM = {ram_gb}GB
$VHDXSize = {vhdx_size}GB

# Verzeichnis erstellen
New-Item -ItemType Directory -Force -Path $VMPath | Out-Null

# VM erstellen
New-VM -Name $VMName -MemoryStartupBytes $RAM -Path $VMPath -Generation 2
Write-Output "VM '$VMName' wurde erstellt."

# VHDX erstellen
$VHDXPath = Join-Path $VMPath "$VMName\\$VMName.vhdx"
New-VHD -Path $VHDXPath -SizeBytes $VHDXSize -Dynamic
Add-VMHardDiskDrive -VMName $VMName -Path $VHDXPath
Write-Output "Virtuelle Festplatte erstellt: $VHDXPath"
"""
        
        if iso_path:
            script += f"""
# ISO einbinden
Add-VMDvdDrive -VMName $VMName -Path '{iso_path}'
Set-VMFirmware -VMName $VMName -FirstBootDevice (Get-VMDvdDrive -VMName $VMName)
Write-Output "ISO eingebunden: {iso_path}"
"""
        
        if self.auto_start_checkbox.isChecked():
            script += f"""
# VM starten
Start-VM -Name $VMName
Write-Output "VM wurde gestartet."
"""
        
        self.log_output(f"Erstelle VM '{vm_name}'...")
        worker = PowerShellWorker(script)
        worker.output_signal.connect(self.log_output)
        worker.error_signal.connect(lambda err: self.log_output(f"FEHLER: {err}"))
        worker.finished_signal.connect(lambda success: self.log_output("✓ VM-Erstellung abgeschlossen!" if success else "✗ VM-Erstellung fehlgeschlagen!"))
        worker.start()
    
    # Post-Installation
    def post_install_config(self):
        vm_name = self.vm_select_combo.currentText()
        if not vm_name:
            QMessageBox.warning(self, "Fehler", "Bitte VM auswählen!")
            return
        
        script = f"$VMName = '{vm_name}'\n"
        
        # Remote-Befehle an die VM
        commands = []
        
        if self.enable_admin_checkbox.isChecked():
            commands.append("net user Administrator /active:yes")
        
        if self.new_username_input.text().strip():
            username = self.new_username_input.text().strip()
            password = self.new_password_input.text().strip()
            commands.append(f"net user {username} {password} /add")
            commands.append(f"net localgroup Administrators {username} /add")
        
        if self.enable_rdp_checkbox.isChecked():
            commands.append("Set-ItemProperty -Path 'HKLM:\\System\\CurrentControlSet\\Control\\Terminal Server' -Name 'fDenyTSConnections' -Value 0")
            commands.append("Enable-NetFirewallRule -DisplayGroup 'Remote Desktop'")
        
        if self.static_ip_checkbox.isChecked() and self.ip_input.text().strip():
            ip = self.ip_input.text().strip()
            gateway = self.gateway_input.text().strip()
            dns = self.dns_input.text().strip()
            commands.append(f"New-NetIPAddress -InterfaceAlias 'Ethernet' -IPAddress {ip} -PrefixLength 24 -DefaultGateway {gateway}")
            if dns:
                commands.append(f"Set-DnsClientServerAddress -InterfaceAlias 'Ethernet' -ServerAddresses {dns}")
        
        if commands:
            script += "Invoke-Command -VMName $VMName -ScriptBlock {\n"
            script += "\n".join(commands)
            script += "\n}"
        
        self.log_output("Führe Post-Installation durch...")
        worker = PowerShellWorker(script)
        worker.output_signal.connect(self.log_output)
        worker.error_signal.connect(lambda err: self.log_output(f"FEHLER: {err}"))
        worker.finished_signal.connect(lambda success: self.log_output("✓ Post-Installation abgeschlossen!" if success else "✗ Post-Installation fehlgeschlagen!"))
        worker.start()
    
    # Active Directory Installation
    def install_ad(self):
        vm_name = self.vm_select_combo.currentText()
        if not vm_name:
            QMessageBox.warning(self, "Fehler", "Bitte VM auswählen!")
            return
        
        domain_name, ok = QMessageBox.getText(self, "Domain-Name", "Bitte Domain-Namen eingeben (z.B. example.local):")
        if not ok or not domain_name:
            return
        
        script = f"""
Invoke-Command -VMName '{vm_name}' -ScriptBlock {{
    # AD DS installieren
    Install-WindowsFeature -Name AD-Domain-Services -IncludeManagementTools
    
    # Domain Controller promoten
    $SafeModePassword = ConvertTo-SecureString 'P@ssw0rd123!' -AsPlainText -Force
    Install-ADDSForest -DomainName '{domain_name}' -SafeModeAdministratorPassword $SafeModePassword -Force
    
    Write-Output "Active Directory wurde installiert. System wird neu gestartet..."
}}
"""
        
        self.log_output("Installiere Active Directory...")
        worker = PowerShellWorker(script)
        worker.output_signal.connect(self.log_output)
        worker.error_signal.connect(lambda err: self.log_output(f"FEHLER: {err}"))
        worker.finished_signal.connect(lambda success: self.log_output("✓ AD-Installation gestartet! VM wird neu gestartet." if success else "✗ AD-Installation fehlgeschlagen!"))
        worker.start()
    
    # AD-Benutzer erstellen
    def create_ad_user(self):
        firstname = self.ad_firstname_input.text().strip()
        lastname = self.ad_lastname_input.text().strip()
        login = self.ad_login_input.text().strip()
        password = self.ad_password_input.text().strip()
        
        if not all([firstname, lastname, login, password]):
            QMessageBox.warning(self, "Fehler", "Bitte alle Felder ausfüllen!")
            return
        
        never_expires = "-PasswordNeverExpires $true" if self.password_never_expires_checkbox.isChecked() else ""
        
        script = f"""
$SecurePassword = ConvertTo-SecureString '{password}' -AsPlainText -Force
New-ADUser -GivenName '{firstname}' -Surname '{lastname}' -Name '{firstname} {lastname}' -SamAccountName '{login}' -UserPrincipalName '{login}@domain.local' -AccountPassword $SecurePassword -Enabled $true {never_expires}
Write-Output "Benutzer {login} wurde erstellt."
"""
        
        self.log_output(f"Erstelle Benutzer {login}...")
        worker = PowerShellWorker(script)
        worker.output_signal.connect(self.log_output)
        worker.error_signal.connect(lambda err: self.log_output(f"FEHLER: {err}"))
        worker.finished_signal.connect(lambda success: self.log_output("✓ Benutzer erstellt!" if success else "✗ Fehler beim Erstellen!"))
        worker.start()
    
    # Excel-Import
    def import_users_from_excel(self):

        if not excel_path:

            return
        
        never_expires = "$true" if self.password_never_expires_checkbox.isChecked() else "$false"
        
        script = f"""
# Excel einlesen (benötigt ImportExcel Modul)
if (-not (Get-Module -ListAvailable -Name ImportExcel)) {{
    Install-Module -Name ImportExcel -Force -Scope CurrentUser
}}
Import-Module ImportExcel

$users = Import-Excel -Path '{excel_path}'
foreach ($user in $users) {{
    $SecurePassword = ConvertTo-SecureString $user.Passwort -AsPlainText -Force
    $params = @{{
        GivenName = $user.Vorname
        Surname = $user.Nachname
        Name = "$($user.Vorname) $($user.Nachname)"
        SamAccountName = $user.Login
        UserPrincipalName = "$($user.Login)@domain.local"
        AccountPassword = $SecurePassword
        Enabled = $true
    }}
    if ({never_expires}) {{
        $params['PasswordNeverExpires'] = $true
    }}
    New-ADUser @params
    Write-Output "Benutzer $($user.Login) erstellt."
}}
"""
        
        self.log_output("Importiere Benutzer aus Excel...")
        worker = PowerShellWorker(script)
        worker.output_signal.connect(self.log_output)
        worker.error_signal.connect(lambda err: self.log_output(f"FEHLER: {err}"))
        worker.finished_signal.connect(lambda success: self.log_output("✓ Benutzer importiert!" if success else "✗ Import fehlgeschlagen!"))
        worker.start()
    
    # OU erstellen
    def create_ou(self):
        ou_name = self.ou_name_input.text().strip()
        if not ou_name:
            QMessageBox.warning(self, "Fehler", "Bitte OU-Name eingeben!")
            return
        
        ou_path = self.ou_path_input.text().strip()
        if not ou_path:
            # Wenn kein Pfad angegeben, Domain ermitteln
            script = f"""
$domain = (Get-ADDomain).DistinguishedName
New-ADOrganizationalUnit -Name '{ou_name}' -Path $domain
Write-Output "OU '{ou_name}' wurde erstellt."
"""
        else:
            script = f"""
New-ADOrganizationalUnit -Name '{ou_name}' -Path '{ou_path}'
Write-Output "OU '{ou_name}' wurde erstellt."
"""
        
        self.log_output(f"Erstelle OU '{ou_name}'...")
        worker = PowerShellWorker(script)
        worker.output_signal.connect(self.log_output)
        worker.error_signal.connect(lambda err: self.log_output(f"FEHLER: {err}"))
        worker.finished_signal.connect(lambda success: self.log_output("✓ OU erstellt!" if success else "✗ Fehler beim Erstellen!"))
        worker.start()
    
    # AD-Gruppe erstellen
    def create_ad_group(self):
        group_name = self.group_name_input.text().strip()
        if not group_name:
            QMessageBox.warning(self, "Fehler", "Bitte Gruppenname eingeben!")
            return
        
        scope = self.group_scope_combo.currentText()
        
        script = f"""
New-ADGroup -Name '{group_name}' -GroupScope {scope} -GroupCategory Security
Write-Output "Gruppe '{group_name}' wurde erstellt."
"""
        
        self.log_output(f"Erstelle Gruppe '{group_name}'...")
        worker = PowerShellWorker(script)
        worker.output_signal.connect(self.log_output)
        worker.error_signal.connect(lambda err: self.log_output(f"FEHLER: {err}"))
        worker.finished_signal.connect(lambda success: self.log_output("✓ Gruppe erstellt!" if success else "✗ Fehler beim Erstellen!"))
        worker.start()
    
    # Benutzer zu Gruppe hinzufügen
    def add_user_to_group(self):
        username = self.user_to_add_input.text().strip()
        groupname = self.group_to_add_input.text().strip()
        
        if not all([username, groupname]):
            QMessageBox.warning(self, "Fehler", "Bitte Benutzername und Gruppenname eingeben!")
            return
        
        script = f"""
Add-ADGroupMember -Identity '{groupname}' -Members '{username}'
Write-Output "Benutzer '{username}' wurde zu Gruppe '{groupname}' hinzugefügt."
"""
        
        self.log_output(f"Füge Benutzer '{username}' zu Gruppe '{groupname}' hinzu...")
        worker = PowerShellWorker(script)
        worker.output_signal.connect(self.log_output)
        worker.error_signal.connect(lambda err: self.log_output(f"FEHLER: {err}"))
        worker.finished_signal.connect(lambda success: self.log_output("✓ Benutzer hinzugefügt!" if success else "✗ Fehler beim Hinzufügen!"))
        worker.start()
